cmake_minimum_required(VERSION 3.20)
project(realtime_budget_linux LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
get_filename_component(GEOGRAM_LIBRARY_DIR "${GEOGRAM_LIBRARY}" DIRECTORY)
set(CMAKE_BUILD_RPATH "${GEOGRAM_LIBRARY_DIR}")
foreach(name geogram_memory local_separation exact_mesh_memory incremental_mesh_memory)
    # 精确守卫目标沿用库名，实际源码文件名为native_separation。
    if(name STREQUAL "local_separation")
        set(source native_separation)
    else()
        set(source ${name})
    endif()
    add_library(${name} SHARED ../${source}.cpp)
    # ctypes包装器与C库不能同名，否则Python会把C ABI库当成扩展模块导入。
    if(name STREQUAL "local_separation")
        set_target_properties(${name} PROPERTIES PREFIX "")
    else()
        set_target_properties(${name} PROPERTIES PREFIX "lib")
    endif()
    target_compile_options(${name} PRIVATE -fno-fast-math -ffp-contract=off -frounding-math)
endforeach()
target_include_directories(geogram_memory PRIVATE "${GEOGRAM_SOURCE}/src/lib")
target_compile_definitions(geogram_memory PRIVATE GEO_DYNAMIC_LIBS GEOGRAM_USE_BUILTIN_DEPS)
target_link_libraries(geogram_memory PRIVATE "${GEOGRAM_LIBRARY}")
target_compile_definitions(exact_mesh_memory PRIVATE CGAL_DISABLE_GMP)
target_compile_definitions(incremental_mesh_memory PRIVATE CGAL_DISABLE_GMP)
